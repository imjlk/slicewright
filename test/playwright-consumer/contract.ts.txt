import { expect as baseExpect, type Page } from '@playwright/test';
import { domTarget, regionTarget, type BrowserTarget } from 'slicewright/browser';
import { expect, toMatchUI, toBeStableUI, toMatchUIBaseline, type PlaywrightBaselineOptions, type PlaywrightStabilityOptions } from 'slicewright/playwright';

declare const page: Page;

async function scenario(target: BrowserTarget): Promise<void> {
  const before = await target.capture();
  await target.click();
  await expect(target).toMatchUI({
    changedFrom: before,
    reference: before,
    pixelRegions: [{ name: 'status', bounds: [{ x: 0, y: 0, width: 10, height: 10 }], pixels: { mean: 1, p99: 1, max: 1 } }],
    stableGeometry: { reference: before.measurement, tolerance: 0.1 },
    contentFits: true,
    text: { label: 'Ready' },
  }, { timeout: 1000, intervals: [25, 50] });
  await expect.soft(target).not.toMatchUI({ reference: before });
  await expect.configure({ timeout: 1000 })(target).toMatchUI({ contentFits: true });
  const customExpect = baseExpect.extend({ toMatchUI, toBeStableUI, toMatchUIBaseline });
  await customExpect(target).toMatchUI({ reference: before });
  const stability: PlaywrightStabilityOptions = { duration: 200, timeout: 1000, intervals: [25], pixels: { mean: 0, p99: 0, max: 0 } };
  await expect(target).toBeStableUI(stability);
  await expect.soft(target).not.toBeStableUI();
  await customExpect(target).toBeStableUI(stability);
  const baseline: PlaywrightBaselineOptions = { environment: 'linux-chromium-wide', checks: { contentFits: true }, timeout: 2000 };
  await expect(target).toMatchUIBaseline('panel.slicewright.json', baseline);
  await customExpect.soft(target).not.toMatchUIBaseline('panel.slicewright.json', baseline);
  // @ts-expect-error baseline environment is required
  await expect(target).toMatchUIBaseline('panel.slicewright.json', {});
  // @ts-expect-error callers cannot replace the approved reference
  await expect(target).toMatchUIBaseline('panel.slicewright.json', { environment: 'test', checks: { reference: before } });
  // @ts-expect-error duration is a number
  await expect(target).toBeStableUI({ duration: '200' });
  // @ts-expect-error all three pixel thresholds are required
  await expect(target).toBeStableUI({ pixels: { max: 1 } });
}

void scenario(domTarget(page, { root: page.locator('button') }));
void scenario(regionTarget(page, {
  measure: () => Promise.resolve({ bounds: { x: 0, y: 0, width: 100, height: 50 } }),
}));
