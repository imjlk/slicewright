import type { Page } from 'playwright-core';
import {
  checkSnapshot,
  domTarget,
  loadBrowserBaseline,
  regionTarget,
  saveBrowserBaseline,
  writeBrowserArtifacts,
  writeBrowserReportHtml,
  type BrowserPixelRegion,
  type BrowserMeasurement,
  type BrowserTarget,
} from 'slicewright/browser';

declare const page: Page;
declare const measure: () => Promise<BrowserMeasurement>;

async function scenario(target: BrowserTarget): Promise<void> {
  const before = await target.capture();
  await saveBrowserBaseline('baseline.json', before, {
    approved: true,
    environment: 'consumer-type-contract',
  });
  const baseline = await loadBrowserBaseline('baseline.json', {
    environment: 'consumer-type-contract',
  });
  const pixelRegions: BrowserPixelRegion[] = [{
    name: 'status', bounds: [{ x: 0, y: 0, width: 10, height: 10 }],
    pixels: { mean: 0.01, p99: 0, max: 16 },
  }];
  checkSnapshot(before, { reference: baseline, pixelRegions });
  await target.hover();
  await target.press();
  await target.release();
  await target.click();
  const actual = await target.capture();
  const report = checkSnapshot(actual, {
    changedFrom: baseline,
    focus: { focused: true, visible: true },
    stableGeometry: { reference: baseline.measurement, tolerance: 0.1 },
    contentFits: true,
    text: { label: 'Accept quest' },
  });
  await writeBrowserArtifacts('artifacts', { actual, changedFrom: baseline, report });
  await writeBrowserReportHtml('artifacts', 'report.html');
}

void scenario(domTarget(page, {
  root: page.locator('button'),
  contentBox: page.locator('button .content'),
  content: { label: page.locator('button .label') },
}));
void scenario(regionTarget(page, { measure }));
